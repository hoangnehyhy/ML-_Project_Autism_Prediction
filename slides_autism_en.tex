\section{Problem and pipeline}

\begin{frame}{Autism Spectrum Disorder and early screening}
\begin{itemize}
    \item Autism Spectrum Disorder (ASD) changes how a person communicates,
          behaves, and interacts with others. Signs usually appear in early childhood.
    \item A full clinical diagnosis is slow and costly: it needs trained
          specialists, long interviews, and often months on a waiting list.
    \item A cheap, fast screening step helps families reach support sooner ---
          when that support helps the most.
    \item Screening does not replace a doctor. It only flags who should be seen first.
\end{itemize}
\end{frame}

\begin{frame}{From a questionnaire to a prediction}
\begin{itemize}
    \item AQ-10: ten short yes/no questions about everyday behaviour
          (\texttt{A1}--\texttt{A10}). Each autistic-like answer adds 1 point.
    \item Each person also gives simple facts: age, sex, jaundice at birth,
          family history of autism, and who filled in the form.
    \item A single fixed cut-off on the score is unreliable in the grey zone
          (score 4--6), where ASD and non-ASD cases overlap.
\end{itemize}
\vspace{0.3cm}
Goal: learn from the AQ-10 answers plus the background facts to flag likely
ASD cases accurately and honestly, with no data leakage.
\end{frame}

\begin{frame}{The target-leakage pitfall}
An early baseline scored a fake-perfect ROC-AUC $= 0.9995$. Why?
\[
\text{enc}(c)=\frac{1}{|\{i:x_i=c\}|}\sum_{i:\,x_i=c} y_i
\quad\text{computed on \emph{all} data before CV.}
\]
\begin{itemize}
    \item Encoding a category by its mean label hides the answer $y$ inside the feature.
    \item The model sees the target through that feature $\Rightarrow$ unreal scores.
    \item Fix: compute encodings inside each fold, or just use one-hot.
\end{itemize}
\end{frame}

\section{Data generation}

\begin{frame}{SMOTE --- interpolating new samples}
Goal: add believable new training rows (not blind copies) so the model is robust.
\[
x_{\text{new}} = x_i + \lambda\,(x_i^{\text{nn}} - x_i),\qquad \lambda \sim U(0,1).
\]
\begin{itemize}
    \item Take a sample $x_i$, pick one of its nearest neighbours $x_i^{\text{nn}}$,
          and place a new point on the line between them.
    \item The new point sits between two real points, so it stays realistic.
    \item Applied only on the training fold, never on validation $\Rightarrow$ no leakage.
\end{itemize}
\end{frame}

\begin{frame}{Tabular GAN (PyTorch, from scratch)}
Two small networks compete in a min--max game:
\[
\min_G \max_D \; \mathbb{E}_{x}\!\left[\log D(x)\right]
+ \mathbb{E}_{z}\!\left[\log\big(1-D(G(z))\big)\right].
\]
\begin{columns}
\begin{column}{0.54\textwidth}
\footnotesize
\begin{itemize}
    \item Generator $G$ maps random noise $z$ to fake patient rows; discriminator
          $D$ tries to separate real rows from fake ones.
    \item As $D$ improves, $G$ is pushed to make more realistic rows; the two
          losses settle near an equilibrium.
    \item We sampled +20k train and +2k test rows $\Rightarrow$ a larger, harder,
          more realistic set.
\end{itemize}
\end{column}
\begin{column}{0.46\textwidth}
\begin{center}
\includegraphics[width=\linewidth,height=0.46\paperheight,keepaspectratio]{fig_loss_gan.png}
\end{center}
\end{column}
\end{columns}
\end{frame}

\section{Models}

\begin{frame}{Evaluation protocol}
\begin{itemize}
    \item One unified 23{,}743-row cleaned set; the same
          \texttt{StratifiedKFold(5, shuffle, seed=42)} for every model.
    \item Headline metrics: F1-Score and ROC-AUC (threshold-free); accuracy reported too.
    \item For each model we search a compact grid of configurations and keep the
          \emph{best by ROC-AUC} --- so every model is compared at its best.
    \item One leaderboard, extended as each method is added.
\end{itemize}
\end{frame}

\begin{frame}{Gradient boosting --- the shared engine}
\small
XGBoost, LightGBM, and CatBoost are all additive ensembles of trees built by
functional gradient descent:
\[
F_m(x)=F_{m-1}(x)+\nu\,h_m(x),\qquad h_m \approx -\,\frac{\partial L}{\partial F_{m-1}}.
\]
\begin{itemize}
    \item Each new tree $h_m$ fits the negative gradient of the loss $L$;
          the learning rate $\nu$ shrinks its step.
    \item XGBoost adds a regularized, second-order objective:
    \[
    \mathcal{L}=\sum_i l(y_i,\hat{y}_i)+\sum_k \Omega(f_k),\qquad
    \Omega(f)=\gamma T+\tfrac12\lambda\lVert w\rVert^2 ,
    \]
    penalising the number of leaves $T$ and the leaf weights $w$.
    \item The three models differ in \emph{how} they grow and regularize these trees.
\end{itemize}
\end{frame}

\begin{frame}{CatBoost: ordered boosting and target statistics}
\small
\begin{itemize}
    \item Uses oblivious trees (the same split on a whole level), which keeps it
          fast and well regularized.
    \item For categories it uses ordered target statistics --- a running label
          average over \emph{earlier} rows only:
    \[
    \hat{x}_i=\frac{\sum_{j<i}\mathbb{1}[x_j=x_i]\,y_j+a\,p}{\sum_{j<i}\mathbb{1}[x_j=x_i]+a}.
    \]
    \item Ordered boosting applies the same use-only-the-past rule to the
          gradients, so a row never sees its own label $\Rightarrow$ no prediction shift.
\end{itemize}
\vspace{0.1cm}
{\footnotesize This directly cures the Phase-1 target-leakage problem.}
\end{frame}

\begin{frame}{TabM: parameter-efficient deep ensembling (BatchEnsemble)}
\small
\begin{itemize}
    \item A plain MLP is one network. An ensemble of $k$ MLPs predicts better,
          but costs about $k\times$ to train.
    \item TabM keeps one shared weight matrix $W$ and gives each member a cheap
          rank-1 adapter $(r_k,s_k)$, so the $k$ members cost $\approx$1$\times$:
    \[
    W_k = W \odot (r_k\,s_k^{\!\top}),\qquad
    \hat{p}=\frac{1}{k}\sum_{m=1}^{k}\mathrm{softmax}\big(f_{W_m}(x)\big).
    \]
    \item Each member sees the data a little differently, so their mistakes differ.
    \item Averaging the members cancels noise and lowers variance, giving
          better-calibrated probabilities.
\end{itemize}
\end{frame}

\begin{frame}{LightGBM: histogram and leaf-wise boosting}
\small
\begin{itemize}
    \item Same gradient-boosting objective as XGBoost, but tuned for speed.
    \item Bins each continuous feature into a histogram (e.g.\ 255 buckets), so
          the best split is found by scanning buckets, not every value.
    \item Grows trees leaf-wise (always split the leaf that cuts the loss most),
          reaching lower error with fewer splits.
    \item GOSS and exclusive feature bundling further cut the work per tree
          without changing the loss it fits.
\end{itemize}
\vspace{0.1cm}
{\footnotesize Completes the gradient-boosting trio --- the fastest of the three.}
\end{frame}

\begin{frame}{TabPFN: in-context learning for tables}
\small
\begin{itemize}
    \item A transformer trained \emph{once} on millions of synthetic tables,
          so it learns a general prior over how tabular data behaves.
    \item At use time you do not train at all: feed the whole training set plus a
          test row as context, and it predicts in a single forward pass
          (in-context learning):
    \[
    p\big(y \mid x,\, \mathcal{D}_{\text{train}}\big)\;\approx\;\text{one in-context forward pass}.
    \]
    \item Because it learned a prior over datasets, it approximates the Bayesian
          posterior predictive --- like averaging over many plausible models.
\end{itemize}
\vspace{0.1cm}
{\footnotesize Best single model out of the box (F1 0.9275, AUC 0.9810); built for small/medium tables.}
\end{frame}

\begin{frame}{Stacking and blending}
\small
\begin{itemize}
    \item Train several diverse base models with CV and keep their out-of-fold (OOF)
          predictions --- each row is predicted only by models that did not see it,
          so it stays leakage-free.
    \item Blending --- average the probabilities: $\;\hat{p}=\tfrac{1}{M}\sum_m p_m(x)$.
    \item Stacking --- a logistic-regression meta-learner reads those
          predictions and learns how much to trust each model:
    \[
    \hat{p}=\sigma\!\Big(\sum_m w_m\,p_m(x)+b\Big).
    \]
    \item The base models make different mistakes, so combining them cancels errors
          and is steadier than any single model.
\end{itemize}
\vspace{0.1cm}
{\footnotesize Base learners: TabPFN + CatBoost + LightGBM + XGBoost.}
\end{frame}

\begin{column}{0.45\textwidth}
\small
A SHAP value is a feature's fair contribution (a Shapley value):
\[
\phi_i\!=\!\!\!\sum_{S\subseteq F\setminus\{i\}}\!\!\!
\frac{|S|!\,(|F|\!-\!|S|\!-\!1)!}{|F|!}\big(f(S\!\cup\!\{i\})\!-\!f(S)\big)
\]
\begin{itemize}
    \item AQ-10 items (A6, A5, A7, A9) sit on top.
    \item Developmental milestones (\texttt{age\_first\_sentence}) matter just as much.
\end{itemize}
\end{column}
\end{columns}
